\section{Ψευδοκώδικας Αλγορίθμων}
\label{sec:app-pseudo}

Οι τρεις δειγματολήπτες Langevin συνοψίζονται παρακάτω σε μορφή ψευδοκώδικα.
Η πλήρης Python υλοποίηση βρίσκεται στους φακέλους
\path{samplers/overdamped/ula/}, \path{samplers/overdamped/mala/} και
\path{samplers/underdamped/kinetic/splitting/baoab/} αντίστοιχα. Το σύνολο διαγνωστικών
είναι υλοποιημένο στο \path{metrics/comprehensive_diagnostics.py}.

\subsection*{ULA --- Unadjusted Langevin Algorithm}

\begin{algorithmic}[1]
\Require βήμα $\eta > 0$, αριθμός βημάτων $N$, κλίση $\nabla U$, αρχικό $x_0$
\State $x \gets x_0$
\For{$k = 1, \ldots, N$}
  \State $\xi \sim \mathcal{N}(0, I_d)$
  \State $x \gets x - \eta\, \nabla U(x) + \sqrt{2\eta}\,\xi$
  \State Καταγραφή $X_k \gets x$
\EndFor
\end{algorithmic}

\subsection*{MALA --- Metropolis-Adjusted Langevin Algorithm}

\begin{algorithmic}[1]
\Require βήμα $\eta > 0$, $N$, $U$ και $\nabla U$, αρχικό $x_0$
\State $x \gets x_0$; $g \gets \nabla U(x)$; $a \gets 0$
\For{$k = 1, \ldots, N$}
  \State $\xi \sim \mathcal{N}(0, I_d)$
  \State $x' \gets x - \eta\, g + \sqrt{2\eta}\,\xi$; \quad $g' \gets \nabla U(x')$
  \State $\log r \gets U(x) - U(x') + \frac{1}{4\eta}\bigl(\|x' - x + \eta g\|^2 - \|x - x' + \eta g'\|^2\bigr)$
  \State $u \sim \mathrm{Uniform}(0,1)$
  \If{$\log u < \log r$}
    \State $x \gets x'$; $g \gets g'$; $a \gets a + 1$
  \EndIf
  \State Καταγραφή $X_k \gets x$
\EndFor
\State Ποσοστό αποδοχής $\gets a / N$
\end{algorithmic}

\subsection*{BAOAB --- Underdamped Langevin (Splitting Ολοκληρωτής)}

\begin{algorithmic}[1]
\Require βήμα $\eta$, τριβή $\gamma$, $N$, $\nabla U$, αρχικό $(x_0, v_0)$
\State $x, v \gets x_0, v_0$
\State $a \gets \exp(-\gamma\eta)$, $s \gets \sqrt{1 - a^2}$; $g \gets \nabla U(x)$
\For{$k = 1, \ldots, N$}
  \State $v \gets v - \tfrac{\eta}{2}\, g$ \Comment{B}
  \State $x \gets x + \tfrac{\eta}{2}\, v$ \Comment{A}
  \State $\xi \sim \mathcal{N}(0, I_d)$; $v \gets a\, v + s\, \xi$ \Comment{O}
  \State $x \gets x + \tfrac{\eta}{2}\, v$ \Comment{A}
  \State $g \gets \nabla U(x)$; $v \gets v - \tfrac{\eta}{2}\, g$ \Comment{B}
  \State Καταγραφή $X_k \gets x$
\EndFor
\end{algorithmic}

\section{Σύνολο Διαγνωστικών}
\label{sec:app-metrics}

Παραθέτουμε τους υπολογισμούς διαγνωστικών του
Κεφαλαίου~\ref{chap:experiments}, όπως υλοποιούνται στο
\path{metrics/comprehensive_diagnostics.py}.

\paragraph{Αποτελεσματικό μέγεθος δείγματος (ESS).}
Για κάθε συντεταγμένη $k$, υπολογίζεται η εμπειρική αυτοσυσχέτιση
$\hat\rho(\ell)$ μέσω FFT, έως lag $L = \min(N/2, 500)$. Ξεκινώντας με
$\hat\tau = 1$, συσσωρεύεται $\hat\tau \gets \hat\tau + 2\hat\rho(\ell)$ για
$\ell = 1, 2, \ldots$, έως το πρώτο lag με $\hat\rho(\ell) < 0.05$ ή έως το $L$.
Έτσι το άθροισμα του~\eqref{eq:iat} κρατά μόνο τους πρώτους όρους. Όταν η
αυτοσυσχέτιση μειώνεται αργά, το άθροισμα σταματά στο $L$ πριν οι όροι γίνουν
μικροί. Τότε το $\hat\tau$ βγαίνει μικρότερο από το σωστό και το ESS μεγαλύτερο. Το ESS ανά συντεταγμένη είναι
$\hat n_\mathrm{eff} = N / \hat\tau$. Το αναφερόμενο \path{ess_mean}
είναι ο μέσος όρος ανά συντεταγμένη. \path{ess_min} και
\path{ess_max} είναι τα ακραία ανά συντεταγμένη.

\paragraph{Quantile MAE.}
Για κάθε συντεταγμένη $k$ και κάθε επίπεδο $q \in \{0.025, 0.5, 0.975\}$,
υπολογίζεται το εμπειρικό $q$-ποσοστημόριο $\hat F_k^{-1}(q)$ μέσω
\texttt{numpy.percentile} και το θεωρητικό $q$-ποσοστημόριο
$F_k^{-1}(q)$ μέσω της αναλυτικής αντίστροφης CDF της περιθώριας κατανομής.
Αναφέρεται ο μέσος όρος ανά συντεταγμένη του μέσου ανά $q$ του
$|\hat F_k^{-1}(q) - F_k^{-1}(q)|$.

\paragraph{Λόγος κάλυψης ουράς.}
Για κάθε συντεταγμένη $k$ και κάθε επίπεδο $q \in \{0.01,\allowbreak\ 0.05,\allowbreak\ 0.95,\allowbreak\
0.99\}$, έστω $\theta = F_k^{-1}(q)$. Υπολογίζεται
\[
  c = \frac{1}{N}\sum_{t=1}^N \mathbf{1}\{X_t^{(k)} \leq \theta\} \quad (q < 0.5),
  \qquad
  c = \frac{1}{N}\sum_{t=1}^N \mathbf{1}\{X_t^{(k)} \geq \theta\} \quad (q > 0.5).
\]
Η αναμενόμενη τιμή του $c$ υπό ακριβή δειγματοληψία είναι
$\min(q, 1-q)$. Αναφέρεται ο λόγος $c / \min(q, 1-q)$.

\paragraph{Στατιστική Kolmogorov--Smirnov.}
Για κάθε συντεταγμένη $k$, υπολογίζεται
$D_n^{(k)} = \sup_x |\hat F_k(x) - F_k(x)|$ χρησιμοποιώντας
\texttt{scipy.stats.kstest} έναντι της αναλυτικής CDF περιθώριας κατανομής.
Αναφέρεται ο μέσος όρος ανά συντεταγμένη ως \path{ks_stat_mean}.
Διατηρείται μόνο η στατιστική του ελέγχου. Η αντίστοιχη $p$-τιμή δεν ερμηνεύεται
ορθά παρουσία αυτοσυσχέτισης και δεν αναφέρεται.

\paragraph{Ποσοστό απόκλισης.}
Κλάσμα δειγμάτων (μετά το burn-in) για τα οποία οποιαδήποτε συντεταγμένη
υπερβαίνει το $10^6$ σε απόλυτη τιμή. Χρησιμοποιείται ως έλεγχος αριθμητικής
σταθερότητας.

\paragraph{Μεροληψίες ροπών.}
$\left|\hat\mu_k - \mu_k\right|$ και $\left|\hat\sigma_k^2 -
\sigma_k^2\right|$ κατά μέσο όρο ανά συντεταγμένη, όπου $\mu_k$ και
$\sigma_k^2$ είναι η αναλυτική μέση τιμή και διασπορά της περιθώριας
κατανομής. Για την κατανομή Cauchy αυτές είναι αόριστες και αναφέρονται ως
\texttt{NaN}.

\section{Αποτελέσματα Μελέτης Σύγκλισης}
\label{sec:app-convergence-table}

Ο πλήρης πίνακας της μελέτης σύγκλισης στο $d=20$ βρίσκεται στο
\path{experiments/results/convergence_d20_20261003_161426.xlsx}
στο φύλλο \path{convergence_summary} (γραμμές ανά εκτέλεση) και στο αρχείο
\path{_agg.xlsx} (διάμεσοι και IQR ανά σημείο ελέγχου), για $21$ διαμορφώσεις
(αλγόριθμος, κατανομή) με $10$ εκτελέσεις η καθεμία.
